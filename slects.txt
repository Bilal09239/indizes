WITH cte_name AS(
    SELECT first_name, COUNT(*) AS count
    FROM PERSON
    GROUP BY first_name 
    ORDER BY count DESC
)

SELECT MIN(count), MAX(count), AVG(count)
FROM cte_name ;




-- Ohne INDEX 
SELECT * FROM PERSON WHERE first_name = 'Michael';
